

--01 — Row Counts
SELECT 'BRONZE_PATIENT' AS TABLE_NAME, COUNT(*) AS ROW_COUNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT
UNION ALL
SELECT 'BRONZE_VISIT', COUNT(*)
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_VISIT
UNION ALL
SELECT 'BRONZE_DIAGNOSIS', COUNT(*)
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_DIAGNOSIS
UNION ALL
SELECT 'BRONZE_MEDICATION', COUNT(*)
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_MEDICATION

UNION ALL

SELECT 'BRONZE_LAB_RESULT', COUNT(*)
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_LAB_RESULT
UNION ALL
SELECT 'BRONZE_BILLING', COUNT(*)
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_BILLING
UNION ALL
SELECT 'BRONZE_INSURANCE', COUNT(*)
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_INSURANCE;



--02 — NULL Validation
SELECT
    COUNT(*) AS TOTAL_ROWS,
    COUNT_IF(PATIENT_ID IS NULL) AS NULL_PATIENT_ID,
    COUNT_IF(DATE_OF_BIRTH IS NULL) AS NULL_DOB,
    COUNT_IF(SOURCE_SYSTEM IS NULL) AS NULL_SOURCE,
    COUNT_IF(INGESTION_TIMESTAMP IS NULL) AS NULL_INGESTION
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT;


-- 03 — Duplicate Validation
SELECT
    PATIENT_ID,
    COUNT(*) AS CNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT
GROUP BY PATIENT_ID
HAVING COUNT(*) > 1;

SELECT
    VISIT_ID,
    COUNT(*) AS CNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_VISIT
GROUP BY VISIT_ID
HAVING COUNT(*) > 1;


--04 — Referential Integrity
SELECT COUNT(*) AS ORPHAN_VISITS
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_VISIT v
LEFT JOIN PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT p
    ON v.PATIENT_ID = p.PATIENT_ID
WHERE p.PATIENT_ID IS NULL;


--05 — Date Validation
--For patient:
SELECT *
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT
WHERE DATE_OF_BIRTH > CURRENT_DATE();
--Insurance:
SELECT *
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_INSURANCE
WHERE EXPIRATION_DATE < EFFECTIVE_DATE;
--Billing:
SELECT *
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_BILLING
WHERE BILLING_DATE > CURRENT_TIMESTAMP();
--Lab:
SELECT *
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_LAB_RESULT
WHERE TEST_DATE > CURRENT_DATE();


--06 — Business Validation
--billing:
SELECT
    BILLING_STATUS,
    COUNT(*) AS CNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_BILLING
GROUP BY BILLING_STATUS
ORDER BY CNT DESC;
--Visit:
SELECT
    VISIT_STATUS,
    COUNT(*) AS CNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_VISIT
GROUP BY VISIT_STATUS
ORDER BY CNT DESC;
--Medication:
SELECT
    STATUS,
    COUNT(*) AS CNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_MEDICATION
GROUP BY STATUS
ORDER BY CNT DESC;
--Lab:
SELECT
    RESULT_STATUS,
    COUNT(*) AS CNT
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_LAB_RESULT
GROUP BY RESULT_STATUS
ORDER BY CNT DESC;


--07 — Bronze → Staging Validation
SELECT
    (SELECT COUNT(*)
     FROM PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT) AS BRONZE_COUNT,

    (SELECT COUNT(*)
     FROM PATIENT_RECORD_DB.STAGING.STG_PATIENT) AS STAGING_COUNT;


--08 — Staging → Silver Validation
SELECT
    COUNT(*) AS FACT_VISITS
FROM PATIENT_RECORD_DB.SILVER.FACT_VISIT;


--09 — Silver → Gold Validation
--DEPARTMENT_PERFORMANCE:
SELECT
    DEPARTMENT_NAME,
    TOTAL_VISITS
FROM PATIENT_RECORD_DB.GOLD.DEPARTMENT_PERFORMANCE
ORDER BY TOTAL_VISITS DESC;

SELECT
    DEPARTMENT_NAME,
    COUNT(*) AS EXPECTED_VISIT_COUNT
FROM PATIENT_RECORD_DB.SILVER.FACT_VISIT
GROUP BY DEPARTMENT_NAME
ORDER BY EXPECTED_VISIT_COUNT DESC;


--10. "PASS / FAIL" validation framework
SELECT
    'PATIENT_ID_NULL_CHECK' AS TEST_NAME,
    CASE
        WHEN COUNT_IF(PATIENT_ID IS NULL) = 0
        THEN 'PASS'
        ELSE 'FAIL'
    END AS STATUS,

    COUNT_IF(PATIENT_ID IS NULL) AS ACTUAL,

    0 AS EXPECTED
FROM PATIENT_RECORD_DB.BRONZE.BRONZE_PATIENT;
