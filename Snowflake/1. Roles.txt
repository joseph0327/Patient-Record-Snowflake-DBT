USE ROLE ACCOUNTADMIN;


-- Create Roles
CREATE OR REPLACE ROLE PATIENT_DATA_ENGINEER;

CREATE OR REPLACE ROLE PATIENT_DATA_ANALYST;

CREATE OR REPLACE ROLE PATIENT_DOCTOR;

CREATE OR REPLACE ROLE PATIENT_AUDITOR;

CREATE OR REPLACE ROLE PATIENT_SECURITY_ADMIN;

SHOW ROLES;



-- Revoking all previlidges
CREATE ROLE IF NOT EXISTS PATIENT_RECORD_ROLE;

REVOKE ALL PRIVILEGES ON ACCOUNT FROM ROLE PATIENT_RECORD_ROLE;

DROP ROLE IF EXISTS PATIENT_RECORD_ROLE;
