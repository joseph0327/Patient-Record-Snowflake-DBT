USE ROLE ACCOUNTADMIN;


-- ============================================================
-- 1. WAREHOUSE ACCESS
-- ============================================================

-- Data Engineer
GRANT USAGE
ON WAREHOUSE WH_PATIENT_ETL
TO ROLE PATIENT_DATA_ENGINEER;

-- Analyst
GRANT USAGE
ON WAREHOUSE WH_PATIENT_ANALYTICS
TO ROLE PATIENT_DATA_ANALYST;

GRANT USAGE
ON WAREHOUSE WH_PATIENT_REPORTING
TO ROLE PATIENT_DATA_ANALYST;


-- ============================================================
-- 2. DATABASE ACCESS
-- ============================================================

-- Data Engineer
GRANT USAGE
ON DATABASE PATIENT_RECORD_DB
TO ROLE PATIENT_DATA_ENGINEER;

-- Analyst
GRANT USAGE
ON DATABASE PATIENT_RECORD_DB
TO ROLE PATIENT_DATA_ANALYST;

-- Auditor
GRANT USAGE
ON DATABASE PATIENT_RECORD_DB
TO ROLE PATIENT_AUDITOR;

-- Security Admin
GRANT USAGE
ON DATABASE PATIENT_RECORD_DB
TO ROLE PATIENT_SECURITY_ADMIN;


-- ============================================================
-- 3. BRONZE ACCESS
-- ============================================================

-- Raw data should primarily be accessible to the
-- data engineering pipeline.

GRANT USAGE
ON SCHEMA PATIENT_RECORD_DB.BRONZE
TO ROLE PATIENT_DATA_ENGINEER;

GRANT SELECT
ON ALL TABLES IN SCHEMA PATIENT_RECORD_DB.BRONZE
TO ROLE PATIENT_DATA_ENGINEER;

GRANT INSERT
ON ALL TABLES IN SCHEMA PATIENT_RECORD_DB.BRONZE
TO ROLE PATIENT_DATA_ENGINEER;

GRANT SELECT
ON FUTURE TABLES IN SCHEMA PATIENT_RECORD_DB.BRONZE
TO ROLE PATIENT_DATA_ENGINEER;

GRANT INSERT
ON FUTURE TABLES IN SCHEMA PATIENT_RECORD_DB.BRONZE
TO ROLE PATIENT_DATA_ENGINEER;


-- ============================================================
-- 4. STAGING ACCESS
-- ============================================================

GRANT USAGE
ON SCHEMA PATIENT_RECORD_DB.STAGING
TO ROLE PATIENT_DATA_ENGINEER;

GRANT SELECT, INSERT, UPDATE, DELETE
ON ALL TABLES IN SCHEMA PATIENT_RECORD_DB.STAGING
TO ROLE PATIENT_DATA_ENGINEER;

GRANT SELECT, INSERT, UPDATE, DELETE
ON FUTURE TABLES IN SCHEMA PATIENT_RECORD_DB.STAGING
TO ROLE PATIENT_DATA_ENGINEER;

-- 5. SNAPSHOT ACCESS

-- ============================================================

TO ROLE PATIENT_DATA_ENGINEER;

GRANT SELECT, INSERT, UPDATE, DELETE
ON ALL TABLES IN SCHEMA PATIENT_RECORD_DB.SILVER
TO ROLE PATIENT_DATA_ENGINEER;

GRANT SELECT, INSERT, UPDATE, DELETE
ON FUTURE TABLES IN SCHEMA PATIENT_RECORD_DB.SILVER

-- ============================================================
-- 10. SECURITY ADMIN
-- ============================================================

-- Security administrator can manage security objects.
GRANT USAGE
ON DATABASE PATIENT_RECORD_DB
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT USAGE
ON SCHEMA PATIENT_RECORD_DB.SECURITY
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT CREATE MASKING POLICY
ON SCHEMA PATIENT_RECORD_DB.SECURITY
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT CREATE ROW ACCESS POLICY
ON SCHEMA PATIENT_RECORD_DB.SECURITY
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT CREATE AUTHENTICATION POLICY
ON SCHEMA PATIENT_RECORD_DB.SECURITY
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT SELECT
ON TABLE PATIENT_RECORD_DB.SECURITY.DOCTOR_USER_MAPPING
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT USAGE
ON SCHEMA PATIENT_RECORD_DB.SILVER
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT SELECT
ON TABLE PATIENT_RECORD_DB.SILVER.FACT_VISIT
TO ROLE PATIENT_SECURITY_ADMIN;

GRANT APPLY AUTHENTICATION POLICY
TO ROLE PATIENT_SECURITY_ADMIN;


-- ============================================================
-- 11. ROLE HIERARCHY
-- ============================================================

-- SECURITYADMIN can manage the custom security administrator.

GRANT ROLE PATIENT_SECURITY_ADMIN
TO ROLE SECURITYADMIN;

SHOW PARAMETERS IN ACCOUNT;

SELECT
    POLICY_NAME,
    POLICY_KIND,
    REF_COLUMN_NAME,
    REF_ENTITY_NAME
FROM SNOWFLAKE.ACCOUNT_USAGE.POLICY_REFERENCES
WHERE REF_DATABASE_NAME = 'PATIENT_RECORD_DB'
AND REF_SCHEMA_NAME = 'SILVER'
AND REF_ENTITY_NAME = 'DIM_PATIENT'
ORDER BY REF_COLUMN_NAME;
