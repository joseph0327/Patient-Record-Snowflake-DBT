-- ============================================================
-- 4. MASKING POLICY - PATIENT NAME
-- ============================================================
CREATE OR REPLACE MASKING POLICY PATIENT_RECORD_DB.SECURITY.MASK_PATIENT_NAME
AS (VAL VARCHAR)
RETURNS VARCHAR ->
CASE
    WHEN IS_ROLE_IN_SESSION('PATIENT_SECURITY_ADMIN') THEN VAL
    WHEN IS_ROLE_IN_SESSION('PATIENT_DOCTOR') THEN VAL
    WHEN IS_ROLE_IN_SESSION('PATIENT_DATA_ENGINEER') THEN VAL
    ELSE 'MASKED PATIENT'
END;
-- ============================================================
-- 5. MASKING POLICY - SSN
-- ============================================================
CREATE OR REPLACE MASKING POLICY PATIENT_RECORD_DB.SECURITY.MASK_SSN
AS (VAL VARCHAR)
RETURNS VARCHAR ->
CASE
    WHEN IS_ROLE_IN_SESSION('PATIENT_SECURITY_ADMIN') THEN VAL
    ELSE '***-**-****'
END;
-- ============================================================
-- 6. MASKING POLICY - EMAIL
-- ============================================================
CREATE OR REPLACE MASKING POLICY PATIENT_RECORD_DB.SECURITY.MASK_EMAIL
AS (VAL VARCHAR)
RETURNS VARCHAR ->
CASE
    WHEN IS_ROLE_IN_SESSION('PATIENT_SECURITY_ADMIN') THEN VAL
    WHEN IS_ROLE_IN_SESSION('PATIENT_DOCTOR') THEN VAL
    WHEN IS_ROLE_IN_SESSION('PATIENT_DATA_ENGINEER') THEN VAL
    WHEN VAL IS NULL THEN NULL
    ELSE REGEXP_REPLACE(VAL,'^(.).+(@.+)$','\\1***\\2')
END;
-- ============================================================
-- 7. MASKING POLICY - PHONE
-- ============================================================
CREATE OR REPLACE MASKING POLICY PATIENT_RECORD_DB.SECURITY.MASK_PHONE
AS (VAL VARCHAR)
RETURNS VARCHAR ->
CASE
    WHEN IS_ROLE_IN_SESSION('PATIENT_SECURITY_ADMIN') THEN VAL
    WHEN IS_ROLE_IN_SESSION('PATIENT_DOCTOR') THEN VAL
    WHEN IS_ROLE_IN_SESSION('PATIENT_DATA_ENGINEER') THEN VAL
    WHEN VAL IS NULL THEN NULL
    ELSE CONCAT('*******',RIGHT(REGEXP_REPLACE(VAL,'[^0-9]',''),4))
END;
-- ============================================================
-- 8. MASKING POLICY - ADDRESS
-- ============================================================
CREATE OR REPLACE MASKING POLICY PATIENT_RECORD_DB.SECURITY.MASK_ADDRESS
AS (VAL VARCHAR)

    WHEN IS_ROLE_IN_SESSION('PATIENT_DATA_ENGINEER') THEN VAL
    ELSE 'MASKED ADDRESS'
END;
-- ============================================================
-- 9. MASKING POLICY - MEDICAL HISTORY (ARRAY type)
-- ============================================================
CREATE OR REPLACE MASKING POLICY PATIENT_RECORD_DB.SECURITY.MASK_MEDICAL_HISTORY
AS (VAL ARRAY)
RETURNS ARRAY ->
CASE

-- 1. CREATE DOCTOR USER MAPPING TABLE
-- ============================================================
USE ROLE ACCOUNTADMIN;
CREATE OR REPLACE TABLE PATIENT_RECORD_DB.SECURITY.DOCTOR_USER_MAPPING (
    SNOWFLAKE_USER VARCHAR(255) NOT NULL,
    DOCTOR_ID VARCHAR(255) NOT NULL,
    ACCESS_ACTIVE BOOLEAN DEFAULT TRUE,
    CREATED_AT TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP(),
    UPDATED_AT TIMESTAMP_NTZ DEFAULT CURRENT_TIMESTAMP(),
    CONSTRAINT UK_DOCTOR_USER_MAPPING UNIQUE (SNOWFLAKE_USER)
);
-- ============================================================
-- 2. INSERT DOCTOR USER MAPPING
-- ============================================================
INSERT INTO PATIENT_RECORD_DB.SECURITY.DOCTOR_USER_MAPPING (
    SNOWFLAKE_USER,
    DOCTOR_ID,
    ACCESS_ACTIVE
)
VALUES
('DR_TEST_001', 'DOC462', TRUE);
-- ============================================================
-- 3. VERIFY DOCTOR MAPPING
-- ============================================================
SELECT
    SNOWFLAKE_USER,
    DOCTOR_ID,
    ACCESS_ACTIVE
FROM PATIENT_RECORD_DB.SECURITY.DOCTOR_USER_MAPPING
WHERE UPPER(SNOWFLAKE_USER) = UPPER('DR_TEST_001');
-- ============================================================
-- 4. VERIFY DOCTOR TO PATIENT RELATIONSHIP
-- ============================================================
SELECT DISTINCT
    PATIENT_ID,
    DOCTOR_ID
FROM PATIENT_RECORD_DB.SILVER.FACT_VISIT
WHERE DOCTOR_ID IS NOT NULL
AND PATIENT_ID IS NOT NULL
ORDER BY DOCTOR_ID, PATIENT_ID
LIMIT 100;
-- ============================================================
-- 5. VERIFY TEST DOCTOR'S PATIENTS
-- ============================================================
SELECT DISTINCT
    V.DOCTOR_ID,
    V.PATIENT_ID
FROM PATIENT_RECORD_DB.SILVER.FACT_VISIT V
WHERE V.DOCTOR_ID = 'DOC462'
AND V.PATIENT_ID IS NOT NULL
ORDER BY V.PATIENT_ID;
-- ============================================================
-- 6. CREATE ROW ACCESS POLICY
-- IMPORTANT:
-- If the policy already exists and is attached to DIM_PATIENT,
-- DO NOT RUN CREATE OR REPLACE.
-- Use the DROP/RECREATE process shown in STEP 7-9.
-- ============================================================
-- ============================================================
-- 7. REMOVE EXISTING RLS POLICY FROM DIM_PATIENT
-- ONLY RUN IF RAP_PATIENT_DOCTOR_ACCESS IS ALREADY ATTACHED
-- ============================================================
USE ROLE PATIENT_SECURITY_ADMIN;
ALTER TABLE PATIENT_RECORD_DB.SILVER.DIM_PATIENT
DROP ROW ACCESS POLICY PATIENT_RECORD_DB.SECURITY.RAP_PATIENT_DOCTOR_ACCESS;
-- ============================================================
-- 8. CREATE CORRECTED ROW ACCESS POLICY
-- PATIENT_ID IS USED BECAUSE DIM_PATIENT DOES NOT HAVE DOCTOR_ID
-- ============================================================
CREATE OR REPLACE ROW ACCESS POLICY PATIENT_RECORD_DB.SECURITY.RAP_PATIENT_DOCTOR_ACCESS
AS (PATIENT_ID VARCHAR)
RETURNS BOOLEAN ->
    CASE
        WHEN IS_ROLE_IN_SESSION('PATIENT_SECURITY_ADMIN') THEN TRUE
        WHEN IS_ROLE_IN_SESSION('PATIENT_DOCTOR') THEN
            EXISTS (
                SELECT 1
                FROM PATIENT_RECORD_DB.SECURITY.DOCTOR_USER_MAPPING M
                JOIN PATIENT_RECORD_DB.SILVER.FACT_VISIT V
                    ON M.DOCTOR_ID = V.DOCTOR_ID
                WHERE UPPER(M.SNOWFLAKE_USER) = UPPER(CURRENT_USER())
                  AND M.ACCESS_ACTIVE = TRUE
                  AND V.PATIENT_ID = PATIENT_ID
            )
        ELSE FALSE
    END;
-- ============================================================
-- 9. ATTACH CORRECTED RLS POLICY TO DIM_PATIENT
-- ============================================================
ALTER TABLE PATIENT_RECORD_DB.SILVER.DIM_PATIENT
ADD ROW ACCESS POLICY PATIENT_RECORD_DB.SECURITY.RAP_PATIENT_DOCTOR_ACCESS
ON (PATIENT_ID);

-- 20. VERIFY EXPECTED PATIENTS
-- RUN AS DR_TEST_001
-- This query shows the patients that DOC462 is associated with.
-- ============================================================
SELECT DISTINCT
    V.DOCTOR_ID,
    V.PATIENT_ID
FROM PATIENT_RECORD_DB.SILVER.FACT_VISIT V
JOIN PATIENT_RECORD_DB.SECURITY.DOCTOR_USER_MAPPING M
    ON M.DOCTOR_ID = V.DOCTOR_ID
WHERE UPPER(M.SNOWFLAKE_USER) = UPPER(CURRENT_USER())
AND M.ACCESS_ACTIVE = TRUE
ORDER BY V.PATIENT_ID;
